⚙️ Configuration Google Sheets

Suis ces 4 étapes pour synchroniser tes commandes sur tous tes appareils. C'est gratuit et prend ~10 minutes.

ÉTAPE 1 — Créer le Google Sheet

Va sur sheets.google.com → Crée un nouveau fichier → Nomme-le Sammhy Print Master. Note l'ID dans l'URL (entre /d/ et /edit).

ÉTAPE 2 — Créer le Apps Script

Dans le Sheet : menu Extensions → Apps Script → Efface le code existant → Colle ce code :

const SHEET_NAME = 'Commandes'; function doGet(e) { const action = e.parameter.action; const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME) || SpreadsheetApp.getActiveSpreadsheet().insertSheet(SHEET_NAME); if (action === 'getAll') { const data = sheet.getDataRange().getValues(); if (data.length <= 1) return jsonResponse([]); const orders = data.slice(1).map(row => JSON.parse(row[1] || '{}')); return jsonResponse(orders); } return jsonResponse({ error: 'Unknown action' }); } function doPost(e) { const payload = JSON.parse(e.postData.contents); const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME) || SpreadsheetApp.getActiveSpreadsheet().insertSheet(SHEET_NAME); if (sheet.getLastRow() === 0) { sheet.appendRow(['ID', 'Data', 'Client', 'Status', 'Date']); } if (payload.action === 'save') { const o = payload.order; sheet.appendRow([o.id, JSON.stringify(o), o.client, o.status, o.date]); return jsonResponse({ ok: true }); } if (payload.action === 'update') { const o = payload.order; const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] === o.id) { sheet.getRange(i + 1, 1, 1, 5).setValues([[o.id, JSON.stringify(o), o.client, o.status, o.date]]); return jsonResponse({ ok: true }); } } sheet.appendRow([o.id, JSON.stringify(o), o.client, o.status, o.date]); return jsonResponse({ ok: true }); } if (payload.action === 'delete') { const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] === payload.id) { sheet.deleteRow(i + 1); return jsonResponse({ ok: true }); } } return jsonResponse({ ok: true }); } if (payload.action === 'uploadFile') { const folder = getOrCreateFolder(payload.orderId); const blob = Utilities.newBlob(Utilities.base64Decode(payload.data), payload.mimeType, payload.fileName); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return jsonResponse({ ok: true, url: file.getUrl(), id: file.getId() }); } return jsonResponse({ error: 'Unknown action' }); } function getOrCreateFolder(orderId) { const rootName = 'Sammhy Print Master'; let root; const roots = DriveApp.getFoldersByName(rootName); root = roots.hasNext() ? roots.next() : DriveApp.createFolder(rootName); const subs = root.getFoldersByName(orderId); return subs.hasNext() ? subs.next() : root.createFolder(orderId); } function jsonResponse(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
